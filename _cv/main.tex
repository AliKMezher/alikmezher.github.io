\section{Research interests}
\cvitem{}{Mathematical physics and nonlinear dispersive PDEs; scaling and mean-field limits
of quantum systems; spectral and dynamical problems in quantum field theory;
particle--radiation dynamics, resonance cascades, and radiative Bose--Einstein
condensation; entropy vanishing, purification of quantum states, and local well-posedness.}
